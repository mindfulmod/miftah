سِينْ وَاوْ يَاءْ فَتْحَة كَسْرَة ضَمَّة فَتْحَتَانْ كَسْرَتَانْ ضَمَّتَانْ بَ بِ بُ تَ تِ تُ ثَ ثِ ثُ جَ جِ جُ حَ حِ حُ خَ خِ خُ دَ دِ دُ ذَ ذِ ذُ رَ رِ رُ زَ زِ زُ سَ سِ سُ شَ شِ شُ صَ صِ صُ ضَ ضِ ضُ طَ طِ طُ ظَ ظِ ظُ عَ عِ عُ غَ غِ غُ فَ فِ فُ قَ قِ قُ كَ كِ كُ لَ لِ لُ مَ مِ مُ نَ نِ نُ هَ هِ هُ وَ وِ وُ يَ يِ يُ بَا بِي بُو تَا تِي تُو ثَا ثِي ثُو جَا جِي جُو حَا حِي حُو خَا خِي خُو دَا دِي دُو ذَا ذِي ذُو رَا رِي رُو زَا زِي زُو سَا سِي سُو شَا شِي شُو صَا صِي صُو ضَا ضِي ضُو طَا طِي طُو ظَا ظِي ظُو عَا عِي عُو غَا غِي غُو فَا فِي فُو قَا قِي قُو كَا كِي كُو لَا لِي لُو مَا مِي مُو نَا نِي نُو هَا هِي هُو وَا وِي وُو يَا يِي يُو بً بٍ بٌ تً تٍ تٌ ثً ثٍ ثٌ جً جٍ جٌ حً حٍ حٌ خً خٍ خٌ دً دٍ دٌ ذً ذٍ ذٌ رً رٍ رٌ زً زٍ زٌ سً سٍ سٌ شً شٍ شٌ صً صٍ صٌ ضً ضٍ ضٌ طً طٍ طٌ ظً ظٍ ظٌ عً عٍ عٌ غً غٍ غٌ فً فٍ فٌ قً قٍ قٌ كً كٍ كٌ لً لٍ لٌ مً مٍ مٌ نً نٍ نٌ هً هٍ هٌ وً وٍ وٌ يً يٍ يٌ بَوْ بَيْ تَوْ تَيْ ثَوْ ثَيْ جَوْ جَيْ حَوْ حَيْ خَوْ خَيْ دَوْ دَيْ ذَوْ ذَيْ رَوْ رَيْ زَوْ زَيْ سَوْ سَيْ
